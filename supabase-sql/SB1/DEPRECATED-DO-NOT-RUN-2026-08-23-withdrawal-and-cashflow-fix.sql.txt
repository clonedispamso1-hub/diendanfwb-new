-- Mục tiêu:
--   1. Alias-qualify TOÀN BỘ tham chiếu "id" trong các RPC rút tiền để hết lỗi
--      `column reference "id" is ambiguous`.
--   2. GIỮ NGUYÊN toàn bộ logic tài chính: phí theo admin_site_settings
--      (withdraw_config), chặn đơn pending trùng, trừ xu khi tạo đơn, hoàn xu
--      khi từ chối / hết hạn.
--   3. Thêm public.my_cash_flow() cho thành viên xem: RÚT TIỀN (kèm trạng thái),
--      CHUYỂN XU ĐI, NHẬN XU VỀ — KHÔNG gồm quà tặng bài viết — CHỈ 3 NGÀY GẦN NHẤT.
--   4. Giữ bản ghi audit; KHÔNG xoá vật lý lịch sử rút tiền / chuyển xu.
--   5. Trả lỗi thân thiện, an toàn cho ứng dụng.

create table if not exists public.withdrawal_requests (
  id             uuid primary key default gen_random_uuid(),

  user_id        uuid not null,
  amount         bigint not null check (amount > 0),
  fee            bigint not null default 0,
  net_amount     bigint not null default 0,
  bank_name      text not null,
  bank_account   text not null,
  account_holder text not null,
  status         text not null default 'pending',
  created_at     timestamptz not null default now()
);

alter table public.withdrawal_requests add column if not exists note         text;
alter table public.withdrawal_requests add column if not exists admin_note   text;
alter table public.withdrawal_requests add column if not exists reviewed_by  uuid;
alter table public.withdrawal_requests add column if not exists reviewed_at  timestamptz;
alter table public.withdrawal_requests add column if not exists processed_at timestamptz;

alter table public.withdrawal_requests drop constraint if exists withdrawal_requests_status_check;

  add constraint withdrawal_requests_status_check
  check (status in ('pending','approved','rejected','refunded'));

create index if not exists withdrawal_requests_user_idx
  on public.withdrawal_requests(user_id, created_at desc);
create index if not exists withdrawal_requests_status_idx
  on public.withdrawal_requests(status, created_at desc);

grant select, insert on public.withdrawal_requests to authenticated;
grant all on public.withdrawal_requests to service_role;

  for select to authenticated

  code          text,
  user_id       uuid,

  fee           bigint,
  net_amount    bigint,

create index if not exists withdrawal_audit_wd_idx
  on public.withdrawal_audit_log(withdrawal_id, created_at desc);
create index if not exists withdrawal_audit_user_idx
  on public.withdrawal_audit_log(user_id, created_at desc);

grant all on public.withdrawal_audit_log to service_role;
grant usage, select on sequence public.withdrawal_audit_log_id_seq to service_role;
alter table public.withdrawal_audit_log enable row level security;
-- Không policy cho authenticated: audit chỉ đọc bằng service_role / RPC admin.

create or replace function public.tg_withdrawal_audit()
returns trigger
language plpgsql
security definer
set search_path = public
as $$
begin
  if tg_op = 'INSERT' then
    insert into public.withdrawal_audit_log(
      withdrawal_id, code, user_id, amount, fee, net_amount,
      old_status, new_status, actor_id, note
    ) values (
      new.id, new.code, new.user_id, new.amount, new.fee, new.net_amount,
      null, new.status, auth.uid(), coalesce(new.note, new.admin_note)
    );
    return new;
  end if;

  if new.status is distinct from old.status then
    insert into public.withdrawal_audit_log(
      withdrawal_id, code, user_id, amount, fee, net_amount,
      old_status, new_status, actor_id, note
    ) values (
      new.id, new.code, new.user_id, new.amount, new.fee, new.net_amount,
      old.status, new.status, auth.uid(), coalesce(new.note, new.admin_note)
    );
  end if;
  return new;
end;
$$;

drop trigger if exists withdrawal_audit_trg on public.withdrawal_requests;
create trigger withdrawal_audit_trg
  after insert or update on public.withdrawal_requests
  for each row execute function public.tg_withdrawal_audit();

-- ----------------------------------------------------------------------------
-- 2) create_withdrawal_request — alias-qualify, giữ nguyên logic tài chính
-- ----------------------------------------------------------------------------
drop function if exists public.create_withdrawal_request(bigint, text, text, text);

create or replace function public.create_withdrawal_request(
  p_amount         bigint,
  p_bank_name      text,
  p_bank_account   text,
  p_account_holder text
)
returns table (
  id uuid, code text, amount bigint, fee bigint, net_amount bigint,
  status text, created_at timestamptz
)
language plpgsql
security definer
set search_path = public
as $$
declare
  v_uid         uuid := auth.uid();
  v_bal         bigint;
  v_fee_percent numeric;
  v_min         bigint;
  v_fee         bigint;
  v_net         bigint;
  v_code        text;
begin
  if v_uid is null then
    raise exception 'Bạn cần đăng nhập để rút tiền.';
  end if;

  select coalesce((s.value->>'fee_percent')::numeric, 20),
         coalesce((s.value->>'min_amount')::bigint, 50000)
    into v_fee_percent, v_min
    from public.admin_site_settings s
   where s.key = 'withdraw_config';

  v_fee_percent := coalesce(v_fee_percent, 20);
  v_min         := coalesce(v_min, 50000);

  if coalesce(p_amount, 0) < v_min then
    raise exception 'Số xu rút tối thiểu là % xu.', to_char(v_min, 'FM999,999,999,999');
  end if;

  if btrim(coalesce(p_bank_name, '')) = ''
     or btrim(coalesce(p_bank_account, '')) = ''
     or btrim(coalesce(p_account_holder, '')) = '' then
    raise exception 'Vui lòng nhập đầy đủ thông tin ngân hàng.';
  end if;

  if exists (
    select 1 from public.profiles p
     where p.id = v_uid and coalesce(p.is_banned, false)
  ) then
    raise exception 'Tài khoản đang bị khoá, không thể rút tiền.';
  end if;

  if exists (
    select 1 from public.withdrawal_requests w
     where w.user_id = v_uid and w.status = 'pending'
  ) then
    raise exception 'Bạn còn 1 yêu cầu đang chờ duyệt.';
  end if;

  select coalesce(p.gem_balance, 0) into v_bal
    from public.profiles p
   where p.id = v_uid
   for update;

  if v_bal is null then
    raise exception 'Không tìm thấy ví của bạn.';
  end if;
  if v_bal < p_amount then
    raise exception 'Số dư không đủ để rút.';
  end if;

  v_fee := floor(p_amount * v_fee_percent / 100.0)::bigint;
  v_net := p_amount - v_fee;

  v_code := 'WD' || to_char(now() at time zone 'Asia/Ho_Chi_Minh', 'YYMMDD')
            || upper(substr(md5(random()::text), 1, 5));

  update public.profiles p
     set gem_balance = coalesce(p.gem_balance, 0) - p_amount
   where p.id = v_uid;

  return query
  insert into public.withdrawal_requests as w (
    code, user_id, amount, fee, net_amount,
    bank_name, bank_account, account_holder, status
  ) values (
    v_code, v_uid, p_amount, v_fee, v_net,
    btrim(p_bank_name), btrim(p_bank_account), btrim(p_account_holder), 'pending'
  )
  returning w.id, w.code, w.amount, w.fee, w.net_amount, w.status, w.created_at;
end;
$$;

grant execute on function public.create_withdrawal_request(bigint, text, text, text) to authenticated;

-- ----------------------------------------------------------------------------
-- 3) review_withdrawal_request — alias-qualify; từ chối = hoàn xu
-- ----------------------------------------------------------------------------
drop function if exists public.review_withdrawal_request(uuid, boolean, text);

create or replace function public.review_withdrawal_request(
  p_id      uuid,
  p_approve boolean,
  p_note    text default null
)
returns jsonb
language plpgsql
security definer
set search_path = public
as $$
declare
  v_user   uuid;
  v_amount bigint;
  v_status text;
  v_code   text;
begin
  begin
    perform public._admin_guard();
  exception
    when undefined_function then null; -- môi trường chưa có guard: bỏ qua
  end;

  select w.user_id, w.amount, w.status, w.code
    into v_user, v_amount, v_status, v_code
    from public.withdrawal_requests w
   where w.id = p_id
   for update;

  if v_user is null then
    raise exception 'Không tìm thấy yêu cầu rút tiền.';
  end if;
  if v_status <> 'pending' then
    raise exception 'Yêu cầu này đã được xử lý.';
  end if;

  update public.withdrawal_requests w
     set status       = case when p_approve then 'approved' else 'rejected' end,
         note         = p_note,
         admin_note   = p_note,
         reviewed_by  = auth.uid(),
         reviewed_at  = now(),
         processed_at = now()
   where w.id = p_id;

  if not p_approve then
    update public.profiles p
       set gem_balance = coalesce(p.gem_balance, 0) + v_amount
     where p.id = v_user;
  end if;

  return jsonb_build_object(
    'ok', true,
    'approved', p_approve,
    'code', v_code,
    'amount', v_amount
  );
end;
$$;

grant execute on function public.review_withdrawal_request(uuid, boolean, text) to authenticated;
grant execute on function public.review_withdrawal_request(uuid, boolean, text) to service_role;

create or replace function public.admin_list_withdrawal_requests(p_status text default null)
returns table (
  id uuid, code text, user_id uuid, amount bigint, fee bigint, net_amount bigint,
  bank_name text, bank_account text, account_holder text, status text,
  note text, reviewed_by uuid, reviewed_at timestamptz, created_at timestamptz
)
language plpgsql
stable
security definer
set search_path = public
as $$
begin
  begin
    perform public._admin_guard();
  exception
    when undefined_function then null;
  end;

  return query
  select w.id, w.code, w.user_id, w.amount, w.fee, w.net_amount,
         w.bank_name, w.bank_account, w.account_holder, w.status,

-- ----------------------------------------------------------------------------
-- 5) expire_pending_withdrawals — hoàn xu đơn pending từ trước 00:00 hôm nay
-- ----------------------------------------------------------------------------
create or replace function public.expire_pending_withdrawals()
returns integer
language plpgsql
security definer
set search_path = public
as $$
declare
  r record;
  n integer := 0;
  v_today_start timestamptz := date_trunc(
    'day', now() at time zone 'Asia/Ho_Chi_Minh'
  ) at time zone 'Asia/Ho_Chi_Minh';
begin
  for r in
    select w.id, w.user_id, w.amount, w.code
      from public.withdrawal_requests w
     where w.status = 'pending'
       and w.created_at < v_today_start
     for update
  loop
    update public.profiles p
       set gem_balance = coalesce(p.gem_balance, 0) + r.amount
     where p.id = r.user_id;

    update public.withdrawal_requests w
       set status       = 'refunded',
           note         = coalesce(w.note, 'Hết hạn — hệ thống tự hoàn xu'),
           admin_note   = coalesce(w.admin_note, 'Hết hạn — hệ thống tự hoàn xu'),
           processed_at = now(),
           reviewed_at  = coalesce(w.reviewed_at, now())
     where w.id = r.id;

    begin
      insert into public.notifications(
        user_id, type, kind, entity_type, entity_id, title, message, data, is_read
      ) values (
        r.user_id, 'withdraw_refunded', 'withdraw_refunded', 'withdrawal', r.id::text,
        'Yêu cầu rút tiền đã hết hạn',
        'Yêu cầu rút tiền #' || coalesce(r.code, '') || ' đã hết hạn. Hệ thống đã hoàn lại '
          || to_char(r.amount, 'FM999,999,999,999') || ' xu.',
        jsonb_build_object('withdrawal_id', r.id, 'code', r.code, 'amount', r.amount),
        false
      );
    exception when others then
      null; -- lỗi thông báo không được chặn việc hoàn xu
    end;

    n := n + 1;
  end loop;

  return n;
end;
$$;

grant execute on function public.expire_pending_withdrawals() to service_role;

-- ----------------------------------------------------------------------------
-- 7) my_cash_flow() — DÒNG TIỀN cho thành viên (3 ngày gần nhất)
--    Gồm: rút tiền (kèm trạng thái), chuyển xu đi, nhận xu về.
--    Loại trừ: quà tặng bài viết (có post_id / action_type khác 'transfer').
-- ----------------------------------------------------------------------------
drop function if exists public.my_cash_flow();
drop function if exists public.my_cash_flow(integer);

create or replace function public.my_cash_flow(p_days integer default 3)
returns table (
  kind       text,          -- 'withdraw' | 'transfer_out' | 'transfer_in'
  id         text,
  code       text,
  amount     bigint,
  fee        bigint,
  net_amount bigint,
  status     text,
  note       text,
  created_at timestamptz,
  -- Đối tác giao dịch (người nhận khi chuyển đi / người gửi khi nhận về).
  counterparty_id        uuid,
  counterparty_name      text,
  counterparty_public_id text
)
language plpgsql
stable
security definer
set search_path = public
as $$
declare
  v_uid   uuid := auth.uid();
  v_days  integer := greatest(1, least(coalesce(p_days, 3), 30));
  v_since timestamptz;
begin
  if v_uid is null then
    return; -- chưa đăng nhập: trả danh sách rỗng, không lỗi kỹ thuật
  end if;

  v_since := now() - make_interval(days => v_days);

  return query
  select 'withdraw'::text as kind,
         w.id::text       as id,
         w.code           as code,
         w.amount         as amount,
         w.fee            as fee,
         w.net_amount     as net_amount,
         w.status         as status,
         coalesce(w.note, w.admin_note) as note,
         w.created_at     as created_at,
         null::uuid       as counterparty_id,
         null::text       as counterparty_name,
         null::text       as counterparty_public_id
    from public.withdrawal_requests w
   where w.user_id = v_uid
     and w.created_at >= v_since
  union all
  select case when t.from_id = v_uid then 'transfer_out' else 'transfer_in' end::text,
         t.id::text,
         null::text,
         t.amount,
         0::bigint,
         t.amount,
         'completed'::text,
         t.note,
         t.created_at,
         cp.id,
         nullif(coalesce(cp.full_name, cp.username), ''),
         cp.public_id::text
    from public.gem_transactions t
    left join public.profiles cp
      on cp.id = case when t.from_id = v_uid then t.to_id else t.from_id end
   where (t.from_id = v_uid or t.to_id = v_uid)
     and t.created_at >= v_since
     and t.post_id is null
     and coalesce(t.action_type, 'transfer') = 'transfer'
   order by created_at desc
   limit 300;
exception
  when undefined_table then
    -- Chưa có bảng gem_transactions: vẫn trả phần rút tiền, không vỡ ứng dụng.
    return query
    select 'withdraw'::text, w.id::text, w.code, w.amount, w.fee, w.net_amount,
           w.status, coalesce(w.note, w.admin_note), w.created_at,
           null::uuid, null::text, null::text
      from public.withdrawal_requests w
     where w.user_id = v_uid
       and w.created_at >= v_since
     order by w.created_at desc
     limit 300;
end;
$$;


grant execute on function public.my_cash_flow(integer) to authenticated;
